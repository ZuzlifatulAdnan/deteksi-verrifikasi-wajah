Panduan Integrasi

Integrasikan Face API ke Stack Anda

Python Face API ini berjalan sebagai AI microservice. Backend utama (Express.js) dan aplikasi mobile (Flutter / Android / iOS) tidak perlu tahu cara kerja AI-nya — cukup kirim foto, terima embedding atau hasil verifikasi.

🏗️ Arsitektur Microservice

Flutter / Android / iOS │ kirim foto (multipart) ▼ Express.js ←── Backend Utama (auth, JWT, bisnis) │ │ │ ├──▶ PostgreSQL / Supabase (simpan embedding, presensi) │ │ └──────────▶ Python Face API (AI saja: extract + compare) port 8080 — internal only, tidak terekspos
✅
Keuntungan: Python API tidak perlu dibuka ke internet. Mobile app hanya punya 1 backend URL. Embedding disimpan di database sendiri — bukan di SQLite Python.

Alur Daftar Wajah

1
Mobile kirim foto ke Express.js
multipart/form-data, field photo
2
Express → POST /api/extract-embedding
Python ekstrak 512-dim embedding + liveness check
3
Express simpan embedding ke DB sendiri
Kolom JSONB atau vector(512) di PostgreSQL/Supabase

Alur Presensi (1:N Identification)

1
Mobile kirim foto ke Express.js
Sudah include liveness frame jika dibutuhkan
2
Express → POST /api/extract-embedding
Python return embedding + is_live + antispoof
3
Express query embedding semua karyawan dari DB
Loop → POST /api/compare-embeddings per user, atau pakai pgvector
4
Express catat presensi, balas mobile
Insert ke tabel presensi, return nama + waktu + similarity

🔌 Endpoint Headless (Khusus Integrasi)

Dua endpoint ini tidak menyimpan data ke database Python — semua kontrol ada di backend Anda.

MethodEndpointDeskripsi
POST/api/extract-embeddingEkstrak embedding 512-dim dari foto. Return embedding + liveness + antispoof. Tidak simpan ke DB.
POST/api/compare-embeddingsBandingkan dua embedding. Input: JSON array float[512] × 2. Return similarity + is_match.
POST/api/registerDaftar + simpan ke SQLite Python (opsional, jika tidak mau urus DB sendiri)
POST/api/presensi/masukClock-in otomatis dengan 1:N di database Python

// services/faceService.js
const axios    = require('axios');
const FormData = require('form-data');
const FACE_API = process.env.FACE_API_URL;

async function extractEmbedding(imageBuffer, mime = 'image/jpeg') {
  const form = new FormData();
  form.append('image', imageBuffer, { filename: 'face.jpg', contentType: mime });
  form.append('check_liveness', 'true');
  const { data } = await axios.post(`${FACE_API}/api/extract-embedding`,
    form, { headers: form.getHeaders(), timeout: 15000 });
  if (!data.success) throw new Error(data.error);
  return data;
}

async function compareEmbeddings(emb1, emb2, threshold = 0.30) {
  const form = new FormData();
  form.append('embedding1', JSON.stringify(emb1));
  form.append('embedding2', JSON.stringify(emb2));
  form.append('threshold',   String(threshold));
  const { data } = await axios.post(`${FACE_API}/api/compare-embeddings`,
    form, { headers: form.getHeaders(), timeout: 5000 });
  if (!data.success) throw new Error(data.error);
  return data;
}

module.exports = { extractEmbedding, compareEmbeddings };

// POST /presensi/:tipe  (masuk | keluar)
router.post('/:tipe(masuk|keluar)', upload.single('photo'), async (req, res) => {
  const { tipe } = req.params;
  const ext = await face.extractEmbedding(req.file.buffer);
  if (!ext.is_live)
    return res.status(400).json({ ok: false, error: 'Liveness gagal' });

  const { rows } = await pool.query(
    'SELECT user_id, name, embedding FROM karyawan WHERE is_active=true'
  );

  let best = null, bestSim = 0;
  for (const row of rows) {
    const cmp = await face.compareEmbeddings(ext.embedding, JSON.parse(row.embedding));
    if (cmp.similarity > bestSim) { bestSim = cmp.similarity; best = { ...row, ...cmp }; }
  }
  if (!best?.is_match)
    return res.status(404).json({ ok: false, error: 'Wajah tidak dikenali' });

  await pool.query(
    'INSERT INTO presensi(user_id,tipe,similarity,waktu) VALUES($1,$2,$3,NOW())',
    [best.user_id, tipe, best.similarity]
  );
  res.json({ ok: true, name: best.name, similarity_pct: best.similarity_pct });
});

// lib/services/face_service.dart
import 'dart:io';
import 'package:dio/dio.dart';

class PresensiResult {
  final bool   ok;
  final String? name;
  final double? similarityPct;
  final String? waktu;
  final String? error;

  PresensiResult({
    required this.ok, this.name, this.similarityPct,
    this.waktu, this.error,
  });

  factory PresensiResult.fromJson(Map<String, dynamic> j) => PresensiResult(
    ok:            j['ok'] ?? false,
    name:          j['name'],
    similarityPct: (j['similarity_pct'] as num?)?.toDouble(),
    waktu:         j['waktu'],
    error:         j['error'],
  );
}

class DaftarResult {
  final bool ok;
  final String? message;
  final String? error;
  DaftarResult({required this.ok, this.message, this.error});
  factory DaftarResult.fromJson(Map j) =>
    DaftarResult(ok: j['ok'] ?? false, message: j['message'], error: j['error']);
}

class FaceService {
  static const _base = 'https://your-express-api.com';  // ganti URL Express.js
  static final _dio = Dio(BaseOptions(
    baseUrl:        _base,
    connectTimeout: const Duration(seconds: 10),
    receiveTimeout: const Duration(seconds: 30),
  ));

  // --- Daftar wajah baru ---
  static Future<DaftarResult> daftarWajah({
    required File photo,
    required String userId,
    required String name,
    String? email,
  }) async {
    final form = FormData.fromMap({
      'user_id': userId,
      'name':    name,
      if (email != null) 'email': email,
      'photo': await MultipartFile.fromFile(
        photo.path, filename: 'face.jpg',
        contentType: DioMediaType('image', 'jpeg'),
      ),
    });
    final res = await _dio.post('/karyawan/daftar', data: form);
    return DaftarResult.fromJson(res.data);
  }

  // --- Presensi masuk / keluar ---
  static Future<PresensiResult> presensi({
    required File photo,
    required String tipe,  // 'masuk' | 'keluar'
    String? deviceId,
    String? location,
  }) async {
    final form = FormData.fromMap({
      'photo': await MultipartFile.fromFile(
        photo.path, filename: 'face.jpg',
        contentType: DioMediaType('image', 'jpeg'),
      ),
      if (deviceId != null) 'device_id': deviceId,
      if (location != null) 'location':  location,
    });
    final res = await _dio.post('/presensi/$tipe', data: form);
    return PresensiResult.fromJson(res.data);
  }
}

class _PresensiScreenState extends State<PresensiScreen> {
  bool           _loading = false;
  PresensiResult? _result;
  String?        _selectedTipe = 'masuk';

  Future<void> _jalankanPresensi() async {
    final picked = await ImagePicker().pickImage(
      source: ImageSource.camera,
      preferredCameraDevice: CameraDevice.front,
      imageQuality: 90,
    );
    if (picked == null) return;

    setState(() { _loading = true; _result = null; });
    try {
      final res = await FaceService.presensi(
        photo: File(picked.path),
        tipe:  _selectedTipe!,

      setState(() => _result = res);

      setState(() => _result = PresensiResult(ok: false, error: e.toString()));

import Foundation; import UIKit

struct PresensiResult: Codable {
  let ok: Bool; let name: String?
  let similarity_pct: Double?; let waktu: String?; let error: String?
}

class FaceService {
  static let shared = FaceService()
  private let base = "https://your-express-api.com"

  func presensi(image: UIImage, tipe: String,
               completion: @escaping (Result<PresensiResult, Error>) -> Void) {
    guard let jpeg = image.jpegData(compressionQuality: 0.9) else { return }
    let url = URL(string: "\(base)/presensi/\(tipe)")!
    var req = URLRequest(url: url); req.httpMethod = "POST"
    let b = "Boundary-\(UUID().uuidString)"
    req.setValue("multipart/form-data; boundary=\(b)", forHTTPHeaderField: "Content-Type")
    var body = Data()
    body += "--\(b)\r\nContent-Disposition: form-data; name=\"photo\"; filename=\"face.jpg\"\r\nContent-Type: image/jpeg\r\n\r\n".data(using: .utf8)!
    body += jpeg
    body += "\r\n--\(b)--\r\n".data(using: .utf8)!
    req.httpBody = body
    URLSession.shared.dataTask(with: req) { data, _, err in
      if let e = err { completion(.failure(e)); return }
      let r = try! JSONDecoder().decode(PresensiResult.self, from: data!)
      completion(.success(r))
    }.resume()
  }
}

-- Karyawan — embedding disimpan sebagai JSONB
CREATE TABLE karyawan (
  id          SERIAL PRIMARY KEY,
  user_id     TEXT UNIQUE NOT NULL,
  name        TEXT NOT NULL,
  email       TEXT,
  phone       TEXT,
  embedding   JSONB NOT NULL,   -- float[512] dari /api/extract-embedding
  face_image  TEXT,
  is_active   BOOLEAN DEFAULT true,
  created_at  TIMESTAMPTZ DEFAULT NOW(),
  updated_at  TIMESTAMPTZ
);

-- Presensi
CREATE TABLE presensi (
  id              SERIAL PRIMARY KEY,
  user_id         TEXT REFERENCES karyawan(user_id),
  tipe            TEXT CHECK(tipe IN ('masuk','keluar')),
  similarity      REAL,
  antispoof_score REAL,
  device_id       TEXT,
  location        TEXT,
  ip_address      TEXT,
  waktu           TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_presensi_waktu ON presensi(waktu DESC);
CREATE INDEX idx_presensi_user  ON presensi(user_id, waktu DESC);

⚡ Supabase + pgvector

Supabase mendukung extension pgvector — bandingkan embedding langsung di SQL, jauh lebih cepat untuk 1:N dengan banyak karyawan.

sql — Supabase SQL Editor
CREATE EXTENSION IF NOT EXISTS vector;

ALTER TABLE karyawan
  ADD COLUMN emb_vec vector(512);

-- Index HNSW untuk pencarian cepat
CREATE INDEX ON karyawan
  USING hnsw (emb_vec vector_cosine_ops);
sql
-- Cari wajah paling mirip (1:N langsung di DB, sangat cepat)
SELECT
  user_id, name,
  1 - (emb_vec <=> '[0.021,-0.134,0.072,...]'::vector) AS similarity
FROM karyawan
WHERE is_active = true
  AND 1 - (emb_vec <=> '[...]'::vector) > 0.30  -- threshold
ORDER BY emb_vec <=> '[...]'::vector
LIMIT 1;
